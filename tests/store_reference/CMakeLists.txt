cmake_minimum_required(VERSION 3.22)
project(FlitzFSStoreReference LANGUAGES C CXX)
if(NOT DEFINED FLITZFS_REFERENCE_SOURCE)
  message(FATAL_ERROR "Explicit source-pinned native storage baseline required")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FLITZFS_REFERENCE_SOURCE}" rev-parse HEAD
  RESULT_VARIABLE ref_result OUTPUT_VARIABLE ref_sha OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT ref_result EQUAL 0 OR NOT ref_sha STREQUAL "5c7d7a0a1aea82cbcf3f8d70b1a1d340a00d2f1e")
  message(FATAL_ERROR "Wrong native object-store comparison baseline")
endif()
set(FLITZFS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FLITZFS_WITH_LZ4 OFF CACHE BOOL "" FORCE)
set(FLITZFS_REQUIRE_ZSTD ON CACHE BOOL "" FORCE)
add_subdirectory("${FLITZFS_REFERENCE_SOURCE}" reference EXCLUDE_FROM_ALL)
add_executable(flitzfs_reference_store_comparison ../store_comparison.cpp)
target_link_libraries(flitzfs_reference_store_comparison PRIVATE flitzfs_core)
