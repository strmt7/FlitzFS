if(FLITZFS_WITH_ZSTD)
  find_path(FLITZFS_ZSTD_INCLUDE zstd.h)
  find_library(FLITZFS_ZSTD_LIBRARY NAMES zstd libzstd zstd_static)
  if(FLITZFS_ZSTD_INCLUDE AND FLITZFS_ZSTD_LIBRARY)
    target_include_directories(flitzfs_core PRIVATE ${FLITZFS_ZSTD_INCLUDE})
    target_link_libraries(flitzfs_core PRIVATE ${FLITZFS_ZSTD_LIBRARY})
    target_compile_definitions(flitzfs_core PRIVATE FLITZFS_HAS_ZSTD=1)
    # Static-linking-only custom allocators must be present, not merely a header.
    # They cap every transactional-codec allocation before calling malloc.
    include(CheckCXXSourceCompiles)
    function(flitzfs_check_store_zstd)
      set(CMAKE_REQUIRED_INCLUDES "${FLITZFS_ZSTD_INCLUDE}")
      set(CMAKE_REQUIRED_LIBRARIES "${FLITZFS_ZSTD_LIBRARY}")
      check_cxx_source_compiles("
        #define ZSTD_STATIC_LINKING_ONLY
        #include <zstd.h>
        static_assert(ZSTD_VERSION_NUMBER >= 10504, \"Zstd too old\");
        int main() {
          ZSTD_customMem memory = { nullptr, nullptr, nullptr };
          ZSTD_CCtx* c = ZSTD_createCCtx_advanced(memory);
          ZSTD_DCtx* d = ZSTD_createDCtx_advanced(memory);
          return int(ZSTD_freeCCtx(c) + ZSTD_freeDCtx(d));
        }" FLITZFS_STORE_ZSTD_LINKS)
      set(FLITZFS_STORE_ZSTD_LINKS "${FLITZFS_STORE_ZSTD_LINKS}" PARENT_SCOPE)
    endfunction()
    flitzfs_check_store_zstd()
    if(FLITZFS_STORE_ZSTD_LINKS)
      target_compile_definitions(flitzfs_core PRIVATE FLITZFS_STORE_HAS_ZSTD=1)
    elseif(FLITZFS_REQUIRE_ZSTD)
      message(FATAL_ERROR "Transactional Zstandard requires the reviewed custom-allocation API")
    else()
      message(STATUS "Bounded store Zstandard unavailable: raw/zero/pattern storage only")
    endif()
    message(STATUS "Zstandard enabled")
  elseif(FLITZFS_REQUIRE_ZSTD)
    message(FATAL_ERROR "Zstandard was required but its headers/library were not found")
  else()
    message(STATUS "Zstandard not found: raw-only build")
  endif()
endif()

if(FLITZFS_EXTENT_REFERENCE_SOURCE)
  if(NOT FLITZFS_BUILD_TESTS OR NOT FLITZFS_REQUIRE_ZSTD OR NOT FLITZFS_REQUIRE_LZ4)
    message(FATAL_ERROR "Extent decoder oracle requires tests and mandatory Zstandard/LZ4")
  endif()
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FLITZFS_EXTENT_REFERENCE_SOURCE}" rev-parse HEAD
    RESULT_VARIABLE ref_result OUTPUT_VARIABLE ref_sha OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT ref_result EQUAL 0 OR NOT ref_sha STREQUAL "4e6a1bb0d144611a09dcbc65276faa676e923882")
    message(FATAL_ERROR "FlitzZip oracle checkout must be exactly the reviewed source pin")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FLITZFS_EXTENT_REFERENCE_SOURCE}" diff --quiet HEAD -- src/core
    RESULT_VARIABLE dirty_result)
  if(NOT dirty_result EQUAL 0)
    message(FATAL_ERROR "FlitzZip oracle source must not have tracked modifications")
  endif()
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/ci/extract_extent_codec_oracle.py"
      "${FLITZFS_EXTENT_REFERENCE_SOURCE}/src/core/archive_blocks.cpp"
    RESULT_VARIABLE native_oracle_result OUTPUT_VARIABLE native_oracle_source ERROR_VARIABLE native_oracle_error TIMEOUT 5)
  if(NOT native_oracle_result EQUAL 0)
    message(FATAL_ERROR "Pinned native decoder extraction failed: ${native_oracle_error}")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/extent_codec_reference.cpp" "${native_oracle_source}")
  add_executable(flitzfs_native_oracle tests/test_extent_codec.cpp "${CMAKE_CURRENT_BINARY_DIR}/extent_codec_reference.cpp")
  target_include_directories(flitzfs_native_oracle PRIVATE src "${FLITZFS_EXTENT_REFERENCE_SOURCE}/src")
  target_compile_definitions(flitzfs_native_oracle PRIVATE FLITZFS_NATIVE_ORACLE=1)
  target_link_libraries(flitzfs_native_oracle PRIVATE flitzfs_core)
  flitzfs_test_codecs(flitzfs_native_oracle)
  add_test(NAME pinned_extent_decoder_reference COMMAND flitzfs_native_oracle)

endif()
