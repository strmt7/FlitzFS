cmake_minimum_required(VERSION 3.22)
project(FlitzFS VERSION 0.1.0 LANGUAGES CXX)

option(FLITZFS_WITH_ZSTD "Enable optional CPU Zstandard compression" ON)
option(FLITZFS_BUILD_TESTS "Build bounded unit tests" ON)

add_library(flitzfs_core src/container.cpp src/read_only_file.cpp src/catalog.cpp)
target_include_directories(flitzfs_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(flitzfs_core PUBLIC cxx_std_20)

if(MSVC)
  target_compile_options(flitzfs_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(flitzfs_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

if(FLITZFS_WITH_ZSTD)
  find_path(FLITZFS_ZSTD_INCLUDE zstd.h)
  find_library(FLITZFS_ZSTD_LIBRARY NAMES zstd libzstd)
  if(FLITZFS_ZSTD_INCLUDE AND FLITZFS_ZSTD_LIBRARY)
    target_include_directories(flitzfs_core PRIVATE ${FLITZFS_ZSTD_INCLUDE})
    target_link_libraries(flitzfs_core PRIVATE ${FLITZFS_ZSTD_LIBRARY})
    target_compile_definitions(flitzfs_core PRIVATE FLITZFS_HAS_ZSTD=1)
    message(STATUS "Zstandard enabled")
  else()
    message(STATUS "Zstandard not found: raw-only build")
  endif()
endif()

add_executable(flitzfs src/cli.cpp)
target_link_libraries(flitzfs PRIVATE flitzfs_core)

if(FLITZFS_BUILD_TESTS)
  enable_testing()
  add_executable(flitzfs_tests tests/test_container.cpp)
  target_link_libraries(flitzfs_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_core_test COMMAND flitzfs_tests)
  add_executable(flitzfs_catalog_tests tests/test_catalog.cpp)
  target_link_libraries(flitzfs_catalog_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_catalog_test COMMAND flitzfs_catalog_tests)
  set_tests_properties(bounded_core_test bounded_catalog_test PROPERTIES TIMEOUT 45)
endif()
