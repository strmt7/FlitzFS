cmake_minimum_required(VERSION 3.22)
project(FlitzFS VERSION 0.2.0 LANGUAGES CXX)

option(FLITZFS_WITH_ZSTD "Enable optional CPU Zstandard compression" ON)
option(FLITZFS_BUILD_TESTS "Build bounded unit tests" ON)
option(FLITZFS_REQUIRE_ZSTD "Fail instead of silently using raw-only mode" OFF)
option(FLITZFS_BUILD_WINFSP "Build experimental Windows callbacks (no mount executable)" OFF)
set(FLITZFS_WINFSP_SDK "" CACHE PATH "Explicit WinFsp SDK root: include, lib and bin")
if(FLITZFS_REQUIRE_ZSTD AND NOT FLITZFS_WITH_ZSTD)
  message(FATAL_ERROR "REQUIRE_ZSTD conflicts with WITH_ZSTD=OFF")
endif()

add_library(flitzfs_core src/container.cpp src/read_only_file.cpp src/catalog.cpp src/projection.cpp src/windows_path.cpp src/mount_plan.cpp src/process_budget.cpp)
target_include_directories(flitzfs_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(flitzfs_core PUBLIC cxx_std_20)

if(MSVC)
  target_compile_options(flitzfs_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(flitzfs_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

if(FLITZFS_WITH_ZSTD)
  find_path(FLITZFS_ZSTD_INCLUDE zstd.h)
  find_library(FLITZFS_ZSTD_LIBRARY NAMES zstd libzstd zstd_static)
  if(FLITZFS_ZSTD_INCLUDE AND FLITZFS_ZSTD_LIBRARY)
    target_include_directories(flitzfs_core PRIVATE ${FLITZFS_ZSTD_INCLUDE})
    target_link_libraries(flitzfs_core PRIVATE ${FLITZFS_ZSTD_LIBRARY})
    target_compile_definitions(flitzfs_core PRIVATE FLITZFS_HAS_ZSTD=1)
    message(STATUS "Zstandard enabled")
  elseif(FLITZFS_REQUIRE_ZSTD)
    message(FATAL_ERROR "Zstandard was required but its headers/library were not found")
  else()
    message(STATUS "Zstandard not found: raw-only build")
  endif()
endif()

add_executable(flitzfs src/cli.cpp)
target_link_libraries(flitzfs PRIVATE flitzfs_core)

if(FLITZFS_BUILD_TESTS)
  enable_testing()
  add_executable(flitzfs_tests tests/test_container.cpp)
  target_link_libraries(flitzfs_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_core_test COMMAND flitzfs_tests)
  find_package(Threads REQUIRED)
  add_executable(flitzfs_catalog_tests tests/test_catalog.cpp)
  target_link_libraries(flitzfs_catalog_tests PRIVATE flitzfs_core Threads::Threads)
  add_test(NAME bounded_catalog_test COMMAND flitzfs_catalog_tests)
  add_executable(flitzfs_projection_tests tests/test_projection.cpp)
  target_link_libraries(flitzfs_projection_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_projection_test COMMAND flitzfs_projection_tests)
  add_executable(flitzfs_admission_tests tests/test_admission.cpp)
  target_link_libraries(flitzfs_admission_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_admission_test COMMAND flitzfs_admission_tests)
  add_executable(flitzfs_windows_path_tests tests/test_windows_path.cpp)
  target_link_libraries(flitzfs_windows_path_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_windows_path_test COMMAND flitzfs_windows_path_tests)
  add_executable(flitzfs_mount_plan_tests tests/test_mount_plan.cpp)
  target_link_libraries(flitzfs_mount_plan_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_mount_plan_test COMMAND flitzfs_mount_plan_tests)
  add_executable(flitzfs_process_budget_tests tests/test_process_budget.cpp)
  target_link_libraries(flitzfs_process_budget_tests PRIVATE flitzfs_core)
  add_test(NAME bounded_process_budget_test COMMAND flitzfs_process_budget_tests)
  set_tests_properties(bounded_process_budget_test bounded_mount_plan_test bounded_core_test bounded_catalog_test bounded_projection_test bounded_admission_test bounded_windows_path_test PROPERTIES TIMEOUT 45)
endif()

# The SDK is supplied explicitly. Configuration never downloads or installs it.
if(FLITZFS_BUILD_WINFSP)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native WinFsp callback preview requires 64-bit Windows")
  endif()
  find_path(FLITZFS_WINFSP_INCLUDE winfsp/winfsp.h PATHS "${FLITZFS_WINFSP_SDK}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(FLITZFS_WINFSP_LIBRARY NAMES winfsp-x64 PATHS "${FLITZFS_WINFSP_SDK}/lib" NO_DEFAULT_PATH REQUIRED)
  find_file(FLITZFS_WINFSP_DLL winfsp-x64.dll PATHS "${FLITZFS_WINFSP_SDK}/bin" NO_DEFAULT_PATH REQUIRED)
  add_library(flitzfs_winfsp src/windows/winfsp_readonly.cpp src/windows/winfsp_session.cpp)
  target_include_directories(flitzfs_winfsp PUBLIC src/windows "${FLITZFS_WINFSP_INCLUDE}")
  target_compile_definitions(flitzfs_winfsp PUBLIC NOMINMAX UNICODE _UNICODE)
  target_link_libraries(flitzfs_winfsp PUBLIC flitzfs_core "${FLITZFS_WINFSP_LIBRARY}" advapi32)
  if(MSVC)
    target_compile_options(flitzfs_winfsp PRIVATE /W4 /permissive-)
  endif()
  if(FLITZFS_BUILD_TESTS)
    add_executable(flitzfs_winfsp_tests tests/test_winfsp_callbacks.cpp)
    target_link_libraries(flitzfs_winfsp_tests PRIVATE flitzfs_winfsp Threads::Threads)
    if(FLITZFS_REQUIRE_ZSTD)
      target_compile_definitions(flitzfs_winfsp_tests PRIVATE FLITZFS_TEST_REQUIRES_ZSTD=1)
    endif()
    add_custom_command(TARGET flitzfs_winfsp_tests POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FLITZFS_WINFSP_DLL}" "$<TARGET_FILE_DIR:flitzfs_winfsp_tests>")
    add_executable(flitzfs_winfsp_session_tests tests/test_winfsp_session.cpp)
    target_link_libraries(flitzfs_winfsp_session_tests PRIVATE flitzfs_winfsp)
    add_custom_command(TARGET flitzfs_winfsp_session_tests POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FLITZFS_WINFSP_DLL}" "$<TARGET_FILE_DIR:flitzfs_winfsp_session_tests>")
    add_test(NAME bounded_winfsp_session_test COMMAND flitzfs_winfsp_session_tests)
    add_test(NAME bounded_winfsp_callback_test COMMAND flitzfs_winfsp_tests)
    set_tests_properties(bounded_winfsp_callback_test bounded_winfsp_session_test PROPERTIES TIMEOUT 45)
  endif()
endif()

# Trusted CPU helper containment, independent of the optional WinFsp SDK.
if(WIN32)
  add_library(flitzfs_process src/windows/bounded_process.cpp)
  target_include_directories(flitzfs_process PUBLIC src/windows)
  target_compile_definitions(flitzfs_process PUBLIC NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(flitzfs_process PUBLIC flitzfs_core)
  if(MSVC)
    target_compile_options(flitzfs_process PRIVATE /W4 /permissive-)
  endif()
  if(FLITZFS_BUILD_TESTS)
    add_executable(flitzfs_process_tests tests/test_bounded_process.cpp)
    target_link_libraries(flitzfs_process_tests PRIVATE flitzfs_process)
    add_test(NAME bounded_windows_process_test COMMAND flitzfs_process_tests)
    set_tests_properties(bounded_windows_process_test PROPERTIES TIMEOUT 45)
  endif()
endif()
