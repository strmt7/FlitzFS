cmake_minimum_required(VERSION 3.22)
project(FlitzFS LANGUAGES CXX)

option(FLITZFS_WITH_LZ4 "Enable optional FLZIP native dictionary decoding" ON)
option(FLITZFS_REQUIRE_LZ4 "Fail if native dictionary dependency is unavailable" OFF)

option(FLITZFS_WITH_ZLIB "Enable optional FLZIP Deflate reading through zlib" ON)
option(FLITZFS_REQUIRE_ZLIB "Fail if FLZIP Deflate dependency is unavailable" OFF)
option(FLITZFS_WITH_ZSTD "Enable optional CPU Zstandard compression" ON)
option(FLITZFS_BUILD_TESTS "Build bounded unit tests" ON)
option(FLITZFS_REQUIRE_ZSTD "Fail instead of silently using raw-only mode" OFF)
option(FLITZFS_BUILD_WINFSP "Build experimental Windows callbacks (no mount executable)" OFF)
option(FLITZFS_BUILD_MOUNT_PROBE "Build hosted-CI-only synthetic mount probe (never default CTest)" OFF)
set(FLITZFS_WINFSP_SDK "" CACHE PATH "Explicit WinFsp SDK root: include, lib and bin")
if(FLITZFS_REQUIRE_ZSTD AND NOT FLITZFS_WITH_ZSTD)
  message(FATAL_ERROR "REQUIRE_ZSTD conflicts with WITH_ZSTD=OFF")
endif()

add_library(flitzfs_core src/container.cpp src/read_only_file.cpp src/catalog.cpp src/projection.cpp src/windows_path.cpp src/mount_plan.cpp src/process_budget.cpp src/catalog_manifest.cpp src/flzip_reader.cpp src/flzip_native.cpp src/gpu_batch.cpp src/worker_wire.cpp)
target_include_directories(flitzfs_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(flitzfs_core PUBLIC cxx_std_20)

if(MSVC)
  target_compile_options(flitzfs_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(flitzfs_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

if(FLITZFS_WITH_ZSTD)
  find_path(FLITZFS_ZSTD_INCLUDE zstd.h)
  find_library(FLITZFS_ZSTD_LIBRARY NAMES zstd libzstd zstd_static)
  if(FLITZFS_ZSTD_INCLUDE AND FLITZFS_ZSTD_LIBRARY)
    target_include_directories(flitzfs_core PRIVATE ${FLITZFS_ZSTD_INCLUDE})
    target_link_libraries(flitzfs_core PRIVATE ${FLITZFS_ZSTD_LIBRARY})
    target_compile_definitions(flitzfs_core PRIVATE FLITZFS_HAS_ZSTD=1)
    message(STATUS "Zstandard enabled")
  elseif(FLITZFS_REQUIRE_ZSTD)
    message(FATAL_ERROR "Zstandard was required but its headers/library were not found")
  else()
    message(STATUS "Zstandard not found: raw-only build")
  endif()
endif()

if(FLITZFS_REQUIRE_ZLIB AND NOT FLITZFS_WITH_ZLIB)
  message(FATAL_ERROR "REQUIRE_ZLIB conflicts with WITH_ZLIB=OFF")
endif()
if(FLITZFS_WITH_ZLIB)
  find_package(ZLIB 1.3.1 QUIET)
  if(ZLIB_FOUND)
    target_link_libraries(flitzfs_core PRIVATE ZLIB::ZLIB)
    target_compile_definitions(flitzfs_core PRIVATE FLITZFS_HAS_ZLIB=1)
  elseif(FLITZFS_REQUIRE_ZLIB)
    message(FATAL_ERROR "zlib was required but not found")
  else()
    message(STATUS "zlib not found: FLZIP Deflate explicitly unsupported")
  endif()
endif()
if(FLITZFS_REQUIRE_LZ4 AND NOT FLITZFS_WITH_LZ4)
  message(FATAL_ERROR "REQUIRE_LZ4 conflicts with WITH_LZ4=OFF")
endif()
if(FLITZFS_WITH_LZ4)
  find_path(FLITZFS_LZ4_INCLUDE lz4.h)
  find_library(FLITZFS_LZ4_LIBRARY NAMES lz4 liblz4 lz4_static)
  if(FLITZFS_LZ4_INCLUDE AND FLITZFS_LZ4_LIBRARY)
    target_include_directories(flitzfs_core PRIVATE "${FLITZFS_LZ4_INCLUDE}")
    target_link_libraries(flitzfs_core PRIVATE "${FLITZFS_LZ4_LIBRARY}")
    target_compile_definitions(flitzfs_core PRIVATE FLITZFS_HAS_LZ4=1)
  elseif(FLITZFS_REQUIRE_LZ4)
    message(FATAL_ERROR "LZ4 was required but not found")
  else()
    message(STATUS "LZ4 unavailable: native dictionary stages explicitly unsupported")
  endif()
endif()
# Fixtures use independent encoders; capability expectations follow actual linking.
function(flitzfs_test_codecs target)
  if(FLITZFS_WITH_LZ4 AND FLITZFS_LZ4_INCLUDE AND FLITZFS_LZ4_LIBRARY)
    target_include_directories(${target} PRIVATE "${FLITZFS_LZ4_INCLUDE}")
    target_link_libraries(${target} PRIVATE "${FLITZFS_LZ4_LIBRARY}")
    target_compile_definitions(${target} PRIVATE FLITZFS_TEST_HAS_LZ4=1)
  endif()
  if(FLITZFS_WITH_ZSTD AND FLITZFS_ZSTD_INCLUDE AND FLITZFS_ZSTD_LIBRARY)
    target_include_directories(${target} PRIVATE "${FLITZFS_ZSTD_INCLUDE}")
    target_link_libraries(${target} PRIVATE "${FLITZFS_ZSTD_LIBRARY}")
    target_compile_definitions(${target} PRIVATE FLITZFS_TEST_HAS_ZSTD=1)
  endif()
  if(FLITZFS_WITH_ZLIB AND ZLIB_FOUND)
    target_link_libraries(${target} PRIVATE ZLIB::ZLIB)
    target_compile_definitions(${target} PRIVATE FLITZFS_TEST_HAS_ZLIB=1)
  endif()
endfunction()

set(FLITZFS_FLITZZIP_SOURCE "" CACHE PATH "Explicit pinned FlitzZip checkout for test-only serializer oracle")
if(FLITZFS_FLITZZIP_SOURCE)
  if(NOT FLITZFS_BUILD_TESTS OR NOT FLITZFS_REQUIRE_ZSTD OR NOT FLITZFS_REQUIRE_ZLIB OR NOT FLITZFS_REQUIRE_LZ4)
    message(FATAL_ERROR "Serializer oracle requires tests and mandatory Zstandard/zlib/LZ4")
  endif()
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FLITZFS_FLITZZIP_SOURCE}" rev-parse HEAD
    RESULT_VARIABLE ref_result OUTPUT_VARIABLE ref_sha OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT ref_result EQUAL 0 OR NOT ref_sha STREQUAL "4e6a1bb0d144611a09dcbc65276faa676e923882")
    message(FATAL_ERROR "FlitzZip oracle checkout must be exactly the reviewed source pin")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FLITZFS_FLITZZIP_SOURCE}" diff --quiet HEAD -- src/core
    RESULT_VARIABLE dirty_result)
  if(NOT dirty_result EQUAL 0)
    message(FATAL_ERROR "FlitzZip oracle source must not have tracked modifications")
  endif()
  add_executable(flitzfs_flzip_oracle tests/test_flzip_oracle.cpp
    "${FLITZFS_FLITZZIP_SOURCE}/src/core/archive_index.cpp")
  target_include_directories(flitzfs_flzip_oracle PRIVATE "${FLITZFS_FLITZZIP_SOURCE}/src")
  target_link_libraries(flitzfs_flzip_oracle PRIVATE flitzfs_core)
  flitzfs_test_codecs(flitzfs_flzip_oracle)
  add_test(NAME pinned_flzip_serializer_oracle COMMAND flitzfs_flzip_oracle)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/ci/extract_flzip_native_oracle.py"
      "${FLITZFS_FLITZZIP_SOURCE}/src/core/archive_blocks.cpp"
    RESULT_VARIABLE native_oracle_result OUTPUT_VARIABLE native_oracle_source ERROR_VARIABLE native_oracle_error TIMEOUT 5)
  if(NOT native_oracle_result EQUAL 0)
    message(FATAL_ERROR "Pinned native decoder extraction failed: ${native_oracle_error}")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/flzip_native_reference.cpp" "${native_oracle_source}")
  add_executable(flitzfs_native_oracle tests/test_flzip_native.cpp "${CMAKE_CURRENT_BINARY_DIR}/flzip_native_reference.cpp")
  target_include_directories(flitzfs_native_oracle PRIVATE "${FLITZFS_FLITZZIP_SOURCE}/src")
  target_compile_definitions(flitzfs_native_oracle PRIVATE FLITZFS_NATIVE_ORACLE=1)
  target_link_libraries(flitzfs_native_oracle PRIVATE flitzfs_core)
  flitzfs_test_codecs(flitzfs_native_oracle)
  add_test(NAME pinned_flzip_native_oracle COMMAND flitzfs_native_oracle)
  set_tests_properties(pinned_flzip_serializer_oracle pinned_flzip_native_oracle PROPERTIES TIMEOUT 45)
endif()
